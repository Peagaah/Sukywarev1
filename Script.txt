local function new(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		inst[k] = v
	end
	inst.Parent = parent
	return inst
end

local function corner(inst, radius)
	return new("UICorner", { CornerRadius = UDim.new(0, radius) }, inst)
end

local function stroke(inst, color, thickness)
	return new("UIStroke", {
		Color = color or THEME.line,
		Thickness = thickness or 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, inst)
end

local function tween(inst, time, props)
	TweenService:Create(inst, TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

local EASING = {
	Back = Enum.EasingStyle.Back,
	Quart = Enum.EasingStyle.Quart,
	Quad = Enum.EasingStyle.Quad,
	Sine = Enum.EasingStyle.Sine,
	Exponential = Enum.EasingStyle.Exponential,
}

local function tweenEx(inst, time, props, style, dir)
	local tw = TweenService:Create(inst, TweenInfo.new(time, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
	tw:Play()
	return tw
end

local function lighten(c, amount)
	local a = amount / 255
	return Color3.new(math.min(c.R + a, 1), math.min(c.G + a, 1), math.min(c.B + a, 1))
end

-- Hover genérico: clareia a propriedade `prop` de `target` enquanto o mouse
-- estiver sobre `hitbox` (se não passar hitbox, usa o próprio target).
local function addHover(target, prop, base, hitbox)
	hitbox = hitbox or target
	hitbox.MouseEnter:Connect(function()
		if anim.hover then
			tween(target, 0.12, { [prop] = lighten(base, anim.hoverStrength) })
		end
	end)
	hitbox.MouseLeave:Connect(function()
		tween(target, 0.12, { [prop] = base })
	end)
end

local function isPointer(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch
end

local orderCounter = 0
local function nextOrder()
	orderCounter += 1
	return orderCounter
end

-- Tudo que usa a cor de destaque se registra aqui para mudar ao vivo
local accentBinds = {}
local function bindAccent(inst, prop)
	table.insert(accentBinds, { inst = inst, prop = prop })
	inst[prop] = ACCENT
end

local function setAccent(color)
	ACCENT = color
	for i = #accentBinds, 1, -1 do
		local b = accentBinds[i]
		if not b.inst:IsDescendantOf(game) then
			table.remove(accentBinds, i)
		else
			tween(b.inst, 0.25, { [b.prop] = color })
		end
	end
end

		Position = UDim2.fromOffset(8, 0),
		BackgroundTransparency = 1,
		Text = text,

		TextTransparency = 1,
		TextXAlignment = Enum.TextXAlignment.Left,
		Font = Enum.Font.GothamMedium,
		TextSize = 13,
	}, toast)
	tween(toast, 0.2, { BackgroundTransparency = 0 })
	tween(st, 0.2, { Transparency = 0 })
	tween(label, 0.2, { TextTransparency = 0 })
	task.delay(2.5, function()
		tween(toast, 0.3, { BackgroundTransparency = 1 })
		tween(st, 0.3, { Transparency = 1 })
		tween(label, 0.3, { TextTransparency = 1 })
		task.wait(0.35)
		toast:Destroy()
	end)
end

local tabs = {}
local buildingTab -- nome da aba que está sendo montada (usado pelo Save/Load)

local function selectTab(name, instant)
	local changed = currentTab ~= name
	currentTab = name
	for tabName, t in pairs(tabs) do
		local active = (tabName == name)
		if active then
			if changed and anim.tabFade and not instant then
				t.page.GroupTransparency = 1
				t.page.Position = UDim2.fromOffset(0, 8)
				t.page.Visible = true
				tweenEx(t.page, anim.tabFadeTime, {
					GroupTransparency = 0,
					Position = UDim2.fromOffset(0, 0),
				}, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
			else
				t.page.GroupTransparency = 0
				t.page.Position = UDim2.fromOffset(0, 0)
				t.page.Visible = true
			end
		else
			t.page.Visible = false
		end
		tween(t.button, 0.15, { TextColor3 = active and THEME.text or THEME.muted })
		if active then
			local x = (t.button.AbsolutePosition.X - header.AbsolutePosition.X) / uiScale.Scale
			local w = t.button.AbsoluteSize.X / uiScale.Scale
			tween(underline, 0.2, { Position = UDim2.new(0, x + 10, 1, -3), Size = UDim2.fromOffset(math.max(w - 20, 4), 2) })
		end
	end
end

	tabCount += 1
	buildingTab = name
	local button = new("TextButton", {
		Size = UDim2.new(0, 0, 1, 0),
		AutomaticSize = Enum.AutomaticSize.X,
		BackgroundTransparency = 1,
		Text = name,
		TextColor3 = THEME.muted,
		Font = Enum.Font.GothamMedium,
		TextSize = 14,
		AutoButtonColor = false,
		LayoutOrder = tabCount,
		ZIndex = 4,
	}, tabBar)
	new("UIPadding", { PaddingLeft = UDim.new(0, 12), PaddingRight = UDim.new(0, 12) }, button)

		if anim.hover and currentTab ~= name then
			tween(button, 0.12, { TextColor3 = lighten(THEME.muted, 60) })
		end
	end)
	button.MouseLeave:Connect(function()
		if currentTab ~= name then
			tween(button, 0.12, { TextColor3 = THEME.muted })
		end
	end)

	-- CanvasGroup: permite o fade de todos os elementos da aba de uma vez
	local page = new("CanvasGroup", {

		BorderSizePixel = 0,

	local right = createColumn(page, UDim2.fromScale(0.5, 0), UDim2.fromScale(0.5, 1))

	tabs[name] = { button = button, page = page }
	button.MouseButton1Click:Connect(function()
		selectTab(name)
	end)
	return left, right
end

----------------------------------------------------------------------
-- REGISTRO PARA SAVE/LOAD
-- Cada componente criado dentro de uma aba (menos a Macro) se registra
-- aqui com a chave "Aba/Texto". O Save/Load percorre esta lista.
----------------------------------------------------------------------
local saveRegistry, saveOrder = {}, {}
local function register(kind, text, widget)
	if not buildingTab or buildingTab == "Macro" then
		return widget
	end
	local base = buildingTab .. "/" .. text
	local key, n = base, 1
	while saveRegistry[key] do
		n += 1
		key = base .. "#" .. n
	end
	saveRegistry[key] = { kind = kind, w = widget }
	table.insert(saveOrder, key)
	return widget
end

function UI.toggle(col, text, default, callback, hint, perm)
	local on = default or false
	local row = new("TextButton", {
		Size = UDim2.new(1, 0, 0, 30),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
		LayoutOrder = nextOrder(),
	}, col)

	local ring = new("Frame", {
		Size = UDim2.fromOffset(20, 20),
		Position = UDim2.new(0, 0, 0.5, -10),
		BackgroundColor3 = THEME.input,
		BorderSizePixel = 0,
	}, row)
	corner(ring, 10)
	local ringStroke = stroke(ring, THEME.off, 1.5)
	addHover(ringStroke, "Color", THEME.off, row)
	addHover(ring, "BackgroundColor3", THEME.input, row)

	local dot = new("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = on and UDim2.fromOffset(12, 12) or UDim2.fromOffset(0, 0),
		BorderSizePixel = 0,
	}, ring)
	corner(dot, 6)
	bindAccent(dot, "BackgroundColor3")

	local label = rowLabel(row, text, 30)
	label.Position = UDim2.fromOffset(30, 0)
	label.Size = UDim2.new(1, -90, 1, 0)

	if hint then
		new("TextLabel", {
			Size = UDim2.new(0, 60, 1, 0),
			Position = UDim2.new(1, -60, 0, 0),
			BackgroundTransparency = 1,
			Text = hint,
			TextColor3 = THEME.muted,
			Font = Enum.Font.Gotham,
			TextSize = 12,
			TextXAlignment = Enum.TextXAlignment.Right,
		}, row)
	end

	local function set(value, silent)
		if value and perm and not hasPerm(perm) and not silent then
			notify("Sem permissão: " .. text)
			return
		end
		on = value
		tween(dot, 0.15, { Size = on and UDim2.fromOffset(12, 12) or UDim2.fromOffset(0, 0) })
		if callback and not silent then
			task.spawn(callback, on)
		end
	end
	row.MouseButton1Click:Connect(function()
		set(not on)
	end)
	return register("toggle", text, { Set = set, Get = function() return on end })
end

	local fill = new("Frame", { Size = UDim2.fromScale(0, 1), BorderSizePixel = 0 }, track)
	corner(fill, 3)
	bindAccent(fill, "BackgroundColor3")
	local knob = new("Frame", {
		Size = UDim2.fromOffset(12, 12),
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0, 0.5),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
	}, track)
	-- círculo perfeito em qualquer tamanho (a bolinha cresce no hover)
	new("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)

	local function render()
		local rel = (value - min) / (max - min)
		fill.Size = UDim2.fromScale(rel, 1)
		knob.Position = UDim2.fromScale(rel, 0.5)
		valueLabel.Text = string.format("%g", value)
	end

	local function apply(newValue, silent)
		newValue = math.clamp(newValue, min, max)
		if newValue ~= value then
			value = newValue
			render()
			if callback and not silent then
				task.spawn(callback, value)
			end
		end
	end

	local function setFromX(x)
		local rel = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		apply(math.floor((min + (max - min) * rel) / step + 0.5) * step)
	end

	local hovering, dragging = false, false

	local function refreshKnob()
		local big = anim.hover and (hovering or dragging)
		tween(knob, 0.1, { Size = big and UDim2.fromOffset(16, 16) or UDim2.fromOffset(12, 12) })
		tween(track, 0.1, { BackgroundColor3 = big and lighten(THEME.off, anim.hoverStrength) or THEME.off })
	end

	hit.MouseEnter:Connect(function()
		hovering = true
		refreshKnob()
	end)
	hit.MouseLeave:Connect(function()
		hovering = false
		refreshKnob()
	end)
	hit.InputBegan:Connect(function(input)
		if isPointer(input) then
			dragging = true
			setFromX(input.Position.X)
			refreshKnob()
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if dragging and isPointer(input) then
			dragging = false
			refreshKnob()
		end
	end)

	render()
	return register("slider", text, { Set = apply, Get = function() return value end })
end

function UI.playerList(col, callback)
	local holder = new("Frame", {
		Size = UDim2.new(1, 0, 0, 100),
		BackgroundColor3 = THEME.input,
		BorderSizePixel = 0,
		LayoutOrder = nextOrder(),
	}, col)
	corner(holder, 8)
	stroke(holder, THEME.line, 1)

	local list = new("ScrollingFrame", {
		Size = UDim2.new(1, -8, 1, -8),
		Position = UDim2.fromOffset(4, 4),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 2,
		ScrollBarImageColor3 = THEME.off,
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		CanvasSize = UDim2.new(),
	}, holder)
	new("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.Name }, list)

	local selected
	local entries = {}

	local function refresh()
		for _, e in pairs(entries) do
			e.button:Destroy()
		end
		table.clear(entries)
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= player then
				local b = new("TextButton", {
					Name = p.Name,
					Size = UDim2.new(1, 0, 0, 28),
					BackgroundColor3 = THEME.row,
					Text = p.Name,
					TextColor3 = THEME.text,
					Font = Enum.Font.GothamMedium,
					TextSize = 13,
					AutoButtonColor = false,
				}, list)
				corner(b, 6)
				addHover(b, "BackgroundColor3", THEME.row)
				local st = stroke(b, ACCENT, 1.5)
				bindAccent(st, "Color")
				st.Transparency = (selected == p) and 0 or 1
				entries[p] = { button = b, stroke = st }
				b.MouseButton1Click:Connect(function()
					if selected and entries[selected] then
						entries[selected].stroke.Transparency = 1
					end
					if selected == p then
						selected = nil
					else
						selected = p
						st.Transparency = 0
					end
					if callback then
						task.spawn(callback, selected)
					end
				end)
			end
		end

	Players.PlayerRemoving:Connect(function(p)
		if selected == p then
			selected = nil
			if callback then
				task.spawn(callback, nil)
			end
		end
		task.defer(refresh)
	end)
	refresh()
	return { Get = function() return selected end }
end

function UI.listBox(col, height, onSelect)
	local holder = new("Frame", {
		Size = UDim2.new(1, 0, 0, height),
		BackgroundColor3 = THEME.input,
		BorderSizePixel = 0,
		LayoutOrder = nextOrder(),
	}, col)
	corner(holder, 8)
	stroke(holder, THEME.line, 1)

	local list = new("ScrollingFrame", {
		Size = UDim2.new(1, -8, 1, -8),
		Position = UDim2.fromOffset(4, 4),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 2,
		ScrollBarImageColor3 = THEME.off,
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		CanvasSize = UDim2.new(),
	}, holder)
	new("UIListLayout", { Padding = UDim.new(0, 2), SortOrder = Enum.SortOrder.LayoutOrder }, list)

	local items, selected, rows = {}, nil, {}

	local function render()
		for _, r in ipairs(rows) do
			r:Destroy()
		end
		table.clear(rows)
		if selected and selected > #items then
			selected = nil
		end
		if #items == 0 then
			rows[1] = new("TextLabel", {
				Size = UDim2.new(1, 0, 0, 24),
				BackgroundTransparency = 1,
				Text = "Nenhum passo ainda",
				TextColor3 = THEME.muted,
				Font = Enum.Font.Gotham,
				TextSize = 12,
			}, list)
			return
		end
		for i, text in ipairs(items) do
			local isSel = selected == i
			local b = new("TextButton", {
				Size = UDim2.new(1, 0, 0, 24),
				BackgroundColor3 = THEME.row,
				BackgroundTransparency = isSel and 0 or 1,
				Text = "  " .. text,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,
				TextColor3 = isSel and THEME.text or THEME.muted,
				Font = Enum.Font.Gotham,
				TextSize = 12,
				AutoButtonColor = false,
				LayoutOrder = i,
			}, list)
			corner(b, 5)
			b.MouseEnter:Connect(function()
				if anim.hover and selected ~= i then
					tween(b, 0.1, { BackgroundTransparency = 0.6 })
				end
			end)
			b.MouseLeave:Connect(function()
				if selected ~= i then
					tween(b, 0.1, { BackgroundTransparency = 1 })
				end
			end)
			b.MouseButton1Click:Connect(function()
				selected = if selected == i then nil else i
				render()
				if onSelect then
					onSelect(selected)
				end
			end)
			rows[i] = b
		end
	end
	render()

	return {
		Set = function(newItems, sel)
			items, selected = newItems, sel
			render()
		end,
		Get = function()
			return selected
		end,
	}
end

local function num(text, min, max, fallback)
	local n = tonumber(text)
	if not n then
		return fallback
	end
	return math.clamp(n, min, max)
end

local function getChar()
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if hum and root and hum.Health > 0 then
		return char, hum, root
	end
	return nil
end

local function aimPartOf(char)
	if cfg.aimPart == "Tronco" then
		return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
	end
	return char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end

local npcCache, npcCacheAt = {}, 0

local function refreshNpcs()
	if os.clock() - npcCacheAt < 1 then
		return
	end
	npcCacheAt = os.clock()
	table.clear(npcCache)
	for _, d in ipairs(workspace:GetDescendants()) do
		if d:IsA("Humanoid") and d.Health > 0 then
			local model = d.Parent
			if model and model:IsA("Model") and not Players:GetPlayerFromCharacter(model) then
				local part = aimPartOf(model)
				if part then
					table.insert(npcCache, { part = part, hum = d, char = model })
				end
			end
		end
	end
end

local function collectTargets(includePlayers, includeNpcs)
	local list = {}
	if includePlayers then
		for _, other in ipairs(Players:GetPlayers()) do
			local sameTeam = cfg.teamCheck and other.Team ~= nil and other.Team == player.Team
			if other ~= player and not sameTeam then
				local char = other.Character
				local hum = char and char:FindFirstChildOfClass("Humanoid")
				local part = char and aimPartOf(char)
				if hum and hum.Health > 0 and part then
					table.insert(list, { part = part, hum = hum, char = char })
				end
			end
		end
	end
	if includeNpcs then
		refreshNpcs()
		for _, e in ipairs(npcCache) do
			if e.part.Parent and e.hum.Health > 0 then
				table.insert(list, e)
			end
		end
	end
	return list
end

local function fovCenter()
	local camera = workspace.CurrentCamera
	if cfg.fovMode == "V2" then
		return camera.ViewportSize / 2
	end
	return UserInputService:GetMouseLocation()
end

local function isVisible(e)
	local camera = workspace.CurrentCamera
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { player.Character }
	local origin = camera.CFrame.Position
	local result = workspace:Raycast(origin, e.part.Position - origin, params)
	return result == nil or result.Instance:IsDescendantOf(e.char)
end

local function pickTarget(list, needVisible)
	local camera = workspace.CurrentCamera
	local center = fovCenter()
	local _, _, myRoot = getChar()
	local best, bestScore
	for _, e in ipairs(list) do
		local pos, onScreen = camera:WorldToViewportPoint(e.part.Position)
		if onScreen then
			local screenDist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
			if screenDist <= cfg.fovSize / 2 and (not needVisible or not cfg.wallCheck or isVisible(e)) then
				local score
				if cfg.priority == "Menor vida" then
					score = e.hum.Health
				elseif cfg.priority == "Mais perto da mira" then
					score = screenDist
				else
					score = myRoot and (e.part.Position - myRoot.Position).Magnitude or screenDist
				end
				if not bestScore or score < bestScore then
					best, bestScore = e, score
				end
			end
		end
	end
	return best
end

-- O script de arma/skill do SEU jogo chama isto para saber onde mirar:

function hooks.GetSilentAimPart()
	if not hasPerm("Aimbot") or not (cfg.silentPlayers or cfg.silentNpc) then
		return nil
	end
	local best = pickTarget(collectTargets(cfg.silentPlayers, cfg.silentNpc), false)
	return best and best.part or nil
end

		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local part = char and aimPartOf(char)
		if hum and hum.Health > 0 and part then
			target = part
		end
	end
	if not target then
		local best = pickTarget(collectTargets(true, false), true)
		target = best and best.part
	end
	if target then
		local camera = workspace.CurrentCamera
		local goal = CFrame.lookAt(camera.CFrame.Position, target.Position)
		camera.CFrame = camera.CFrame:Lerp(goal, cfg.smooth)
	end
end)

local function makeGlide(g)
	return function()
		if not g.enabled or g.busy then
			return
		end
		local _, hum, root = getChar()
		if not root then
			return
		end
		local dir = hum.MoveDirection
		if dir.Magnitude < 0.1 then
			local look = workspace.CurrentCamera.CFrame.LookVector
			dir = Vector3.new(look.X, 0, look.Z)
		end
		if dir.Magnitude < 0.01 then
			return
		end
		dir = dir.Unit
		g.busy = true
		local endAt = os.clock() + g.time / 100
		local conn
		conn = RunService.Heartbeat:Connect(function()
			local _, _, r = getChar()
			if not r or os.clock() >= endAt then
				conn:Disconnect()
				g.busy = false
				return
			end
			local y = g.keepHeight and 0 or r.AssemblyLinearVelocity.Y
			r.AssemblyLinearVelocity = Vector3.new(dir.X * g.speed, y, dir.Z * g.speed)
		end)
	end
end

local triggerGlideA = makeGlide(cfg.glideA)
local triggerGlideB = makeGlide(cfg.glideB)

-- DASH (CFrame para a frente, na direção da câmera) -------------------
local function doDash()
	local d = cfg.dash
	if not d.enabled or not hasPerm("Glide") then
		return
	end
	local char, _, root = getChar()
	if not root then
		return
	end
	local look = workspace.CurrentCamera.CFrame.LookVector
	local dir = Vector3.new(look.X, 0, look.Z)
	if dir.Magnitude < 0.01 then
		return
	end
	dir = dir.Unit
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { char }
	local hit = workspace:Raycast(root.Position, dir * d.distance, params)
	local travel = hit and math.max(hit.Distance - 3, 0) or d.distance -- não atravessa parede
	root.CFrame = root.CFrame + dir * travel
end

----------------------------------------------------------------------
-- SORU (o servidor move o personagem)
----------------------------------------------------------------------
-- Direção do Soru normal (tecla Q):
--   "Câmera"    = exatamente para onde a câmera aponta (com altura se "pitch" estiver ligado)
--   "Movimento" = para onde você está andando
--   "Mouse"     = para o ponto do mundo sob o mouse
local function getSoruDirection()
	local _, hum, root = getChar()
	if not root then
		return nil
	end
	local camera = workspace.CurrentCamera
	local dir
	if cfg.soru.direction == "Câmera" then
		local l = camera.CFrame.LookVector
		dir = cfg.soru.pitch and l or Vector3.new(l.X, 0, l.Z)
	elseif cfg.soru.direction == "Mouse" then
		local m = UserInputService:GetMouseLocation()
		local ray = camera:ViewportPointToRay(m.X, m.Y)
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = { player.Character }
		local hit = workspace:Raycast(ray.Origin, ray.Direction * 1000, params)
		local pos = hit and hit.Position or (ray.Origin + ray.Direction * 100)
		local d = pos - root.Position
		dir = Vector3.new(d.X, 0, d.Z)
	else
		dir = hum.MoveDirection
		if dir.Magnitude < 0.1 then
			local l = camera.CFrame.LookVector
			dir = Vector3.new(l.X, 0, l.Z)
		end
	end
	if dir.Magnitude < 0.01 then
		return nil
	end
	return dir.Unit
end

local SORU_SOUND_ID = "" -- ex.: "rbxassetid://123456" (vazio = sem som)

-- Rastro em linha (Part neon que some) e/ou em partículas (ParticleEmitter)
local function soruEffects(origin, target, dir, travel)
	local s = cfg.soru
	local mid = origin:Lerp(target, 0.5)
	-- se o Soru for quase vertical, o "cima" do CFrame precisa mudar para não dar erro
	local up = math.abs(dir.Y) > 0.99 and Vector3.xAxis or Vector3.yAxis
	local cf = CFrame.lookAt(mid, target, up)

	if s.trailMode == "Linha" or s.trailMode == "Ambos" then
		local part = Instance.new("Part")
		part.Anchored = true
		part.CanCollide = false
		part.CanQuery = false
		part.Material = Enum.Material.Neon
		part.Color = s.trailColor
		part.Size = Vector3.new(s.trailWidth, s.trailWidth, travel)
		part.CFrame = cf
		part.Parent = workspace
		TweenService:Create(part, TweenInfo.new(s.trailTime), {
			Transparency = 1,
			Size = Vector3.new(0.1, 0.1, travel),
		}):Play()
		Debris:AddItem(part, s.trailTime + 0.1)
	end

	if s.trailMode == "Partículas" or s.trailMode == "Ambos" then
		local box = Instance.new("Part")
		box.Anchored = true
		box.CanCollide = false
		box.CanQuery = false
		box.Transparency = 1
		box.Size = Vector3.new(math.max(s.trailWidth, 1.5), math.max(s.trailWidth, 1.5), travel)
		box.CFrame = cf
		box.Parent = workspace

		local em = Instance.new("ParticleEmitter")
		em.Rate = 0
		em.Shape = Enum.ParticleEmitterShape.Box
		em.ShapeStyle = Enum.ParticleEmitterShapeStyle.Volume
		em.Color = ColorSequence.new(s.trailColor)
		em.LightEmission = 1
		em.Lifetime = NumberRange.new(s.trailTime, s.trailTime * 1.5)
		em.Speed = NumberRange.new(0, 3)
		em.Size = NumberSequence.new({
			NumberSequenceKeypoint.new(0, s.trailWidth * 0.7),
			NumberSequenceKeypoint.new(1, 0),
		})
		em.Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0),
			NumberSequenceKeypoint.new(1, 1),
		})
		em.Parent = box
		em:Emit(math.clamp(math.floor(travel * 2), 10, 200))
		Debris:AddItem(box, s.trailTime * 1.5 + 0.5)
	end
end

local function soruLocal(dir, distance)
	local s = cfg.soru
	local char, _, root = getChar()
	if not root then
		return
	end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { char }

	local origin = root.Position
	local hit = workspace:Raycast(origin, dir * distance, params)
	local travel = hit and math.max(hit.Distance - 3, 0) or distance
	local target = origin + dir * travel

	-- O personagem sempre olha na horizontal (evita girar de cabeça pra baixo)
	local flat = Vector3.new(dir.X, 0, dir.Z)
	if flat.Magnitude > 0.01 then
		root.CFrame = CFrame.lookAt(target, target + flat)
	else
		root.CFrame = CFrame.new(target) * (root.CFrame - root.CFrame.Position)
	end

	-- Soru aéreo contínuo: zera a queda e flutua um instante, sem precisar tocar no chão
	if s.air then
		s.airUntil = os.clock() + s.airHover
		local v = root.AssemblyLinearVelocity
		root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)
	end

	if s.trail and travel > 1 then
		soruEffects(origin, target, dir, travel)
	end
	if s.sound and SORU_SOUND_ID ~= "" then
		local snd = Instance.new("Sound")
		snd.SoundId = SORU_SOUND_ID
		snd.Parent = root
		snd:Play()
		Debris:AddItem(snd, 3)
	end
end

-- Pronto para usar? (Instant Soru ignora o cooldown)
local function soruReady(force)
	local s = cfg.soru
	return force or (s.enabled and (s.instant or os.clock() >= s.readyAt))
end

local function doSoru(force)
	local s = cfg.soru
	if not soruReady(force) then
		return
	end
	local dir = getSoruDirection()
	if not dir then
		return
	end
	if not s.instant then
		s.readyAt = os.clock() + s.cooldown
	end
	soruLocal(dir, s.distance)
end

-- Alvo mais próximo que está dentro da tela (campo de visão da câmera)
local function nearestVisibleTarget()
	local s = cfg.soru
	local camera = workspace.CurrentCamera
	local _, _, myRoot = getChar()
	if not myRoot then
		return nil
	end
	local best, bestDist
	for _, e in ipairs(collectTargets(s.targetPlayers, s.targetNpc)) do
		local _, onScreen = camera:WorldToViewportPoint(e.part.Position)
		if onScreen then
			local d = (e.part.Position - myRoot.Position).Magnitude
			if not bestDist or d < bestDist then
				best, bestDist = e, d
			end
		end
	end
	return best, bestDist
end

local function doSoruTarget()
	local s = cfg.soru
	if not soruReady(false) then
		return
	end
	local e, dist = nearestVisibleTarget()
	if not e then
		notify("Nenhum alvo na tela")
		return
	end
	local _, _, root = getChar()
	if not root then
		return
	end
	local d = e.part.Position - root.Position
	if d.Magnitude < 0.01 then
		return
	end
	if not s.instant then
		s.readyAt = os.clock() + s.cooldown
	end
	-- vai até perto do alvo (para ~4 studs antes), no máximo a distância do Soru
	soruLocal(d.Unit, math.min(s.distance, math.max(dist - 4, 1)))
end

local flyState = { enabled = false, lv = nil, att = nil, vel = Vector3.zero }
local flyToggle

local function stopFly()
	if flyState.lv then
		flyState.lv:Destroy()
		flyState.lv = nil
	end
	if flyState.att then
		flyState.att:Destroy()
		flyState.att = nil
	end
	local _, hum = getChar()
	if hum then
		hum.PlatformStand = false
		hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	end
end

local function startFly()
	local _, hum, root = getChar()
	if not root then
		return false
	end
	stopFly()
	local att = Instance.new("Attachment")
	att.Parent = root
	local lv = Instance.new("LinearVelocity")
	lv.Attachment0 = att
	lv.MaxForce = math.huge
	lv.RelativeTo = Enum.ActuatorRelativeTo.World
	lv.VectorVelocity = Vector3.zero
	lv.Parent = root
	flyState.att, flyState.lv = att, lv
	flyState.vel = Vector3.zero
	hum.PlatformStand = true
	return true
end

RunService.Heartbeat:Connect(function(dt)
	-- FLY suave ---------------------------------------------------------
	if flyState.enabled then
		local _, hum, root = getChar()
		if not root or not flyState.lv or not flyState.lv.Parent then
			flyState.enabled = false
			stopFly()
			if flyToggle then
				flyToggle.Set(false, true)
			end
		else
			local cam = workspace.CurrentCamera.CFrame
			local look, right = cam.LookVector, cam.RightVector

			-- Modo "Teclas": W/S/A/D andam na horizontal; subir/descer só com Espaço/Ctrl
			if cfg.fly.vertical == "Teclas" then
				look = Vector3.new(look.X, 0, look.Z)
				look = look.Magnitude > 0.01 and look.Unit or Vector3.zero
			end
			right = Vector3.new(right.X, 0, right.Z)
			right = right.Magnitude > 0.01 and right.Unit or Vector3.zero

			local move = Vector3.zero
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += look end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= look end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += right end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= right end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.yAxis end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.yAxis end
			if move.Magnitude < 0.01 and hum.MoveDirection.Magnitude > 0.1 then
				move = hum.MoveDirection -- celular / controle
			end

			local goal = move.Magnitude > 0.01 and move.Unit * cfg.fly.speed or Vector3.zero

			-- suavização: quanto maior "smooth", mais devagar acelera/freia
			local rate = 25 - 0.22 * cfg.fly.smooth
			local alpha = cfg.fly.smooth <= 0 and 1 or (1 - math.exp(-rate * dt))
			flyState.vel = (flyState.vel or Vector3.zero):Lerp(goal, alpha)
			flyState.lv.VectorVelocity = flyState.vel

			-- o personagem vira suavemente para onde a câmera olha (sem inclinar)
			local flat = Vector3.new(cam.LookVector.X, 0, cam.LookVector.Z)
			if flat.Magnitude > 0.01 then
				local pos = root.Position
				root.CFrame = root.CFrame:Lerp(CFrame.lookAt(pos, pos + flat), 0.25)
			end
		end
	end

	-- Soru aéreo: segura a queda por alguns instantes depois de cada Soru
	if cfg.soru.air and os.clock() < cfg.soru.airUntil then
		local _, _, root = getChar()
		if root then
			local v = root.AssemblyLinearVelocity
			root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)
		end
	end
end)

local esp = {
	on = false,
	names = false,
	dist = false,
	health = false,
	maxDist = 1000, -- em studs (o slider vai até 20000)
	style = "Contorno", -- "Contorno", "Preenchido" ou "Nenhum"
	ignoreTeam = true,
	box = false, -- Box ESP (contorno do corpo)
	nameMode = "Apelido", -- "Apelido" (DisplayName) ou "Usuário"
	unit = "Metros", -- "Metros" ou "Studs"
	textSize = 13,
	tracers = false, -- linhas da tela até os jogadores
	tracerGlobal = true, -- true = linhas para TODOS, ignorando a distância máxima
	tracerOrigin = "Baixo", -- "Baixo", "Centro" ou "Mouse"
	tracerThick = 1.5,
}
local espObjects = {}
local tracerLines = {}

-- 1 stud = 0,28 metro
local function fmtDist(studs)
	if esp.unit == "Metros" then
		return string.format("%.1f m", studs * 0.28)
	end
	return math.floor(studs) .. " studs"
end

local function clearEsp(p)
	local o = espObjects[p]
	if o then
		for _, inst in pairs(o) do
			if typeof(inst) == "Instance" then
				inst:Destroy()
			end
		end
		espObjects[p] = nil
	end
end

local function clearTracer(p)
	local line = tracerLines[p]
	if line then
		line:Destroy()
		tracerLines[p] = nil
	end
end

local function buildEsp(char)
	local root = char:FindFirstChild("HumanoidRootPart")
	if not root then
		return nil
	end
	local o = { char = char }
	o.hl = new("Highlight", {
		Adornee = char,
		OutlineColor = Color3.new(1, 1, 1),
		DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
	}, gui)
	-- Box ESP: contorno do corpo inteiro (aparece através das paredes)
	o.sel = new("SelectionBox", {
		Adornee = char,
		LineThickness = 0.04,
		SurfaceTransparency = 1,
	}, gui)
	o.bb = new("BillboardGui", {
		Adornee = root,
		Size = UDim2.fromOffset(200, 70),
		StudsOffset = Vector3.new(0, 3.8, 0),
		AlwaysOnTop = true,
	}, gui)
	o.label = new("TextLabel", {
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		TextColor3 = Color3.new(1, 1, 1),
		TextStrokeTransparency = 0.4,
		Font = Enum.Font.GothamBold,
		TextSize = 13,
	}, o.bb)
	return o
end

local lastEsp = 0
RunService.Heartbeat:Connect(function()
	if os.clock() - lastEsp < 0.1 then
		return
	end
	lastEsp = os.clock()

	if not esp.on then
		for p in pairs(espObjects) do
			clearEsp(p)
		end
		return
	end

	local _, _, myRoot = getChar()
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= player then
			local char = p.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			local root = char and char:FindFirstChild("HumanoidRootPart")
			local sameTeam = esp.ignoreTeam and p.Team ~= nil and p.Team == player.Team
			if not (hum and root and hum.Health > 0) or sameTeam then
				clearEsp(p)
			else
				local o = espObjects[p]
				if not o or o.char ~= char then
					clearEsp(p)
					o = buildEsp(char)
					espObjects[p] = o
				end
				if o then
					local dist = myRoot and (root.Position - myRoot.Position).Magnitude or 0
					local inRange = dist <= esp.maxDist
					o.hl.Enabled = inRange and esp.style ~= "Nenhum"
					o.hl.FillColor = ACCENT
					o.hl.FillTransparency = esp.style == "Preenchido" and 0.5 or 1
					o.sel.Visible = inRange and esp.box
					o.sel.Color3 = ACCENT
					o.bb.Enabled = inRange and (esp.names or esp.dist or esp.health)
					o.label.TextSize = esp.textSize
					local lines = {}
					if esp.names then
						table.insert(lines, esp.nameMode == "Apelido" and p.DisplayName or p.Name)
					end
					if esp.dist then
						table.insert(lines, fmtDist(dist))
					end
					if esp.health then
						table.insert(lines, math.floor(hum.Health) .. "/" .. math.floor(hum.MaxHealth) .. " HP")
					end
					o.label.Text = table.concat(lines, "\n")
				end
			end
		end
	end
end)

-- TRACERS: uma linha da tela até CADA jogador, atualizada todo frame ------
RunService.RenderStepped:Connect(function()
	if not (esp.on and esp.tracers) then
		for p in pairs(tracerLines) do
			clearTracer(p)
		end
		return
	end

	local camera = workspace.CurrentCamera
	local screen = camera.ViewportSize
	local origin
	if esp.tracerOrigin == "Centro" then
		origin = screen / 2
	elseif esp.tracerOrigin == "Mouse" then
		origin = UserInputService:GetMouseLocation()
	else
		origin = Vector2.new(screen.X / 2, screen.Y)
	end

	local _, _, myRoot = getChar()
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= player then
			local char = p.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			local root = char and char:FindFirstChild("HumanoidRootPart")
			local sameTeam = esp.ignoreTeam and p.Team ~= nil and p.Team == player.Team
			local show = hum ~= nil and root ~= nil and hum.Health > 0 and not sameTeam
			if show and not esp.tracerGlobal then
				local dist = myRoot and (root.Position - myRoot.Position).Magnitude or 0
				show = dist <= esp.maxDist
			end

			if not show then
				clearTracer(p)
			else
				local line = tracerLines[p]
				if not line then
					line = new("Frame", {
						AnchorPoint = Vector2.new(0.5, 0.5),
						BorderSizePixel = 0,
						ZIndex = 0, -- fica atrás do painel
					}, gui)
					tracerLines[p] = line
				end

				local v = camera:WorldToViewportPoint(root.Position)
				local pt = Vector2.new(v.X, v.Y)
				if v.Z < 0 then
					-- jogador atrás da câmera: espelha para a linha apontar para o lado certo
					pt = origin + (origin - pt)
				end
				pt = Vector2.new(math.clamp(pt.X, 0, screen.X), math.clamp(pt.Y, 0, screen.Y))

				local d = pt - origin
				local len = d.Magnitude
				line.Visible = len > 1
				line.Size = UDim2.fromOffset(len, esp.tracerThick)
				line.Position = UDim2.fromOffset((origin.X + pt.X) / 2, (origin.Y + pt.Y) / 2)
				line.Rotation = math.deg(math.atan2(d.Y, d.X))
				line.BackgroundColor3 = ACCENT
			end
		end
	end
end)

local keyEvent = Instance.new("BindableEvent")
keyEvent.Name = "SukyWareKey"
keyEvent.Parent = game:GetService("ReplicatedStorage")

local function getMouseWorld(maxDist)
	local camera = workspace.CurrentCamera
	local m = UserInputService:GetMouseLocation()
	local ray = camera:ViewportPointToRay(m.X, m.Y)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { player.Character }
	local hit = workspace:Raycast(ray.Origin, ray.Direction * 2000, params)
	if hit then
		return hit.Position, true
	end
	return ray.Origin + ray.Direction * (maxDist or 2000), false
end

local mouseCache, mouseCacheAt = Vector3.zero, 0
local function sendInput(input, down)
	if os.clock() - mouseCacheAt > 0.01 then
		mouseCache = getMouseWorld()
		mouseCacheAt = os.clock()
	end
	keyEvent:Fire(input, down, mouseCache)
	if down and typeof(input) == "EnumItem" and input.EnumType == Enum.KeyCode then
		onKeyDown(input)
	end
end
hooks.SendInput = sendInput

local function teleportToMouse(maxDist)
	local _, _, root = getChar()
	if not root then
		return
	end
	local pos, hit = getMouseWorld(maxDist)
	local d = pos - root.Position
	if d.Magnitude < 1 then
		return
	end
	if d.Magnitude > maxDist then
		d = d.Unit * maxDist
	end
	local target = root.Position + d
	if hit then
		target += Vector3.new(0, 3, 0)
	end
	root.CFrame = CFrame.new(target) * (root.CFrame - root.CFrame.Position)
end

-- TP SPAM -------------------------------------------------------------
local tps = {
	enabled = false,
	key = Enum.KeyCode.T,
	sendKey = Enum.KeyCode.R,
	interval = 10, -- ms entre cada disparo
	mode = "Segurar", -- "Segurar" ou "Alternar"
	action = "Teleporte nativo", -- ou "Enviar tecla (jogo)"
	maxDist = 80,
	active = false,
	accum = 0,
}

local function tpsFire()
	if tps.action == "Enviar tecla (jogo)" then
		sendInput(tps.sendKey, true)
		sendInput(tps.sendKey, false)
	else
		teleportToMouse(tps.maxDist)
	end
end

RunService.Heartbeat:Connect(function(dt)
	if not tps.active then
		tps.accum = 0
		return
	end
	tps.accum += dt * 1000
	local n = math.min(math.floor(tps.accum / tps.interval), 30) -- vários por frame se o intervalo for menor que um frame
	if n >= 30 then
		tps.accum = 0
	else
		tps.accum -= n * tps.interval
	end
	for _ = 1, n do
		tpsFire()
	end
end)

local function tpsInput(input, began)
	if not tps.enabled or input.KeyCode ~= tps.key or not hasPerm("Soru") then
		return
	end
	if tps.mode == "Alternar" then
		if began then
			tps.active = not tps.active
			notify("TP Spam " .. (tps.active and "ligado" or "desligado"))
		end
	else
		tps.active = began
	end
end

-- MACROS --------------------------------------------------------------
local MACRO_COUNT = 5
local macroKeys = { Enum.KeyCode.Z, Enum.KeyCode.X, Enum.KeyCode.C, Enum.KeyCode.V, Enum.KeyCode.B }
local macros = {}
for i = 1, MACRO_COUNT do
	macros[i] = {
		enabled = false,
		key = macroKeys[i],
		mode = "Segurar", -- "Segurar", "Alternar" ou "Uma vez"
		loops = 0, -- 0 = infinito
		cycleDelay = 0, -- ms entre ciclos
		speed = 100, -- % (200 = o dobro da velocidade)
		jitter = 0, -- variação aleatória em ms (+/-)
		steps = {},
		running = false,
		token = 0,
	}
end

local function describeStep(i, st)
	local t = st.type
	local hold, delay = st.hold or 0, st.delay or 0
	if t == "Tecla" then
		return string.format("%d. Tecla %s  (segura %gms, espera %gms)", i, st.key.Name, hold, delay)
	elseif t == "Segurar" then
		return string.format("%d. Segurar %s  (espera %gms)", i, st.key.Name, delay)
	elseif t == "Soltar" then
		return string.format("%d. Soltar %s  (espera %gms)", i, st.key.Name, delay)
	elseif t == "Clique" then
		return string.format("%d. Clique %s  (segura %gms, espera %gms)", i, st.btn == 2 and "dir." or "esq.", hold, delay)
	elseif t == "Esperar" then
		return string.format("%d. Esperar %gms", i, delay)
	elseif t == "Spam" then
		return string.format("%d. Spam %s x%g  (segura %gms, espera %gms)", i, st.key.Name, st.count or 1, hold, delay)
	elseif t == "Soru" then
		return string.format("%d. Soru  (espera %gms)", i, delay)
	elseif t == "TP mouse" then
		return string.format("%d. TP até o mouse  (espera %gms)", i, delay)
	end
	return i .. ". " .. tostring(t)
end

local function runMacro(m, tok, cycles)
	local steps = table.clone(m.steps)
	if #steps == 0 then
		m.running = false
		notify("Esse macro não tem passos")
		return
	end
	local debt = 0
	local held = {}

	local function alive()
		return m.running and m.token == tok
	end

	local function scaled(v)
		v = v * 100 / math.max(m.speed, 1)
		if m.jitter > 0 then
			v += (math.random() * 2 - 1) * m.jitter
		end
		return math.max(v, 0)
	end

	-- Espera em ms. O Roblox só "acorda" a cada frame (~16ms a 60 FPS), então
	-- tempos menores são somados e compensados para a média ficar certa.
	local function sleep(v)
		debt += scaled(v)
		if debt >= 8 then
			local target = os.clock() + debt / 1000
			repeat
				RunService.Heartbeat:Wait()
			until os.clock() >= target or not alive()
			debt = -math.max(os.clock() - target, 0) * 1000
		end
	end

	local function press(input, down)
		sendInput(input, down)
		held[input] = down or nil
	end

	local function tap(input, st)
		press(input, true)
		sleep(st.hold or 0)
		press(input, false)
		sleep(st.delay or 0)
	end

	local function doStep(st)
		local t = st.type
		if t == "Tecla" then
			tap(st.key, st)
		elseif t == "Segurar" then
			press(st.key, true)
			sleep(st.delay or 0)
		elseif t == "Soltar" then
			press(st.key, false)
			sleep(st.delay or 0)
		elseif t == "Clique" then
			tap(st.btn == 2 and Enum.UserInputType.MouseButton2 or Enum.UserInputType.MouseButton1, st)
		elseif t == "Esperar" then
			sleep(st.delay or 0)
		elseif t == "Spam" then
			for _ = 1, st.count or 1 do
				if not alive() then
					break
				end
				tap(st.key, st)
			end
		elseif t == "Soru" then
			doSoru(true)
			sleep(st.delay or 0)
		elseif t == "TP mouse" then
			teleportToMouse(tps.maxDist)
			sleep(st.delay or 0)
		end
	end

	local done = 0
	while alive() do
		for _, st in ipairs(steps) do
			if not alive() then
				break
			end
			doStep(st)
		end
		done += 1
		if cycles > 0 and done >= cycles then
			break
		end
		if not alive() then
			break
		end
		sleep(m.cycleDelay)
		RunService.Heartbeat:Wait() -- nunca trava o jogo, mesmo com todos os tempos em 0
	end

	for input in pairs(held) do
		sendInput(input, false) -- solta qualquer tecla que ficou presa
	end
	if m.token == tok then
		m.running = false
	end
end

local function startMacro(m, cycles)
	if m.running or not hasPerm("Macro") then
		return
	end
	m.running = true
	m.token += 1
	task.spawn(runMacro, m, m.token, cycles)
end

local function stopMacro(m)
	m.running = false
end

-- GRAVAÇÃO ------------------------------------------------------------
local function makeTap(input, hold, delay)
	if typeof(input) == "EnumItem" and input.EnumType == Enum.UserInputType then
		return { type = "Clique", btn = input == Enum.UserInputType.MouseButton2 and 2 or 1, hold = hold, delay = delay }
	end
	return { type = "Tecla", key = input, hold = hold, delay = delay }
end

local function buildStepsFromEvents(events)
	-- 1) eventos -> Segurar / Soltar / Esperar
	local raw = {}
	local last = events[1] and events[1].t or 0
	for _, e in ipairs(events) do
		local gap = math.floor((e.t - last) * 1000 + 0.5)
		if gap > 0 then
			table.insert(raw, { type = "Esperar", delay = gap })
		end
		last = e.t
		table.insert(raw, { type = e.down and "Segurar" or "Soltar", key = e.input, delay = 0 })
	end

	-- 2) Segurar K + Esperar + Soltar K vira um toque (Tecla/Clique)
	local merged, i = {}, 1
	while i <= #raw do
		local a, b, c = raw[i], raw[i + 1], raw[i + 2]
		if a.type == "Segurar" and b and b.type == "Esperar" and c and c.type == "Soltar" and c.key == a.key then
			table.insert(merged, makeTap(a.key, b.delay, 0))
			i += 3
		elseif a.type == "Segurar" and b and b.type == "Soltar" and b.key == a.key then
			table.insert(merged, makeTap(a.key, 0, 0))
			i += 2
		else
			table.insert(merged, a)
			i += 1
		end
	end

	-- 3) Esperar depois de um passo vira o "esperar após" dele
	local out = {}
	for _, st in ipairs(merged) do
		local prev = out[#out]
		if st.type == "Esperar" and prev and prev.type ~= "Esperar" then
			prev.delay = (prev.delay or 0) + st.delay
		else
			table.insert(out, st)
		end
	end
	return out
end

local function inputKey(input)
	return input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode or input.UserInputType
end

local function isRecordable(input)
	if input.UserInputType == Enum.UserInputType.Keyboard then
		local k = input.KeyCode
		if k == toggleKey or k == tps.key then
			return false
		end
		for _, m in ipairs(macros) do
			if m.key == k then
				return false
			end
		end
		return true
	end
	return input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.MouseButton2
end

-- ENTRADAS REAIS (ativar macros, TP Spam e gravar) ---------------------
UserInputService.InputBegan:Connect(function(input, processed)
	if processed then
		return
	end
	if recorder.on then
		if isRecordable(input) then
			local key = inputKey(input)
			recorder.downs[key] = true
			table.insert(recorder.events, { t = os.clock(), down = true, input = key })
		end
		return
	end
	if input.UserInputType ~= Enum.UserInputType.Keyboard then
		return
	end
	tpsInput(input, true)
	for _, m in ipairs(macros) do
		if m.enabled and m.key == input.KeyCode then
			if m.mode == "Alternar" then
				if m.running then
					stopMacro(m)
					notify("Macro desligado")
				else
					startMacro(m, m.loops)
					notify("Macro ligado")
				end
			else
				startMacro(m, m.mode == "Uma vez" and math.max(1, m.loops) or m.loops)
			end
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if recorder.on then
		local key = inputKey(input)
		if recorder.downs[key] then
			recorder.downs[key] = nil
			table.insert(recorder.events, { t = os.clock(), down = false, input = key })
		end
		return
	end
	if input.UserInputType ~= Enum.UserInputType.Keyboard then
		return
	end
	tpsInput(input, false)
	for _, m in ipairs(macros) do
		if m.mode == "Segurar" and m.key == input.KeyCode then
			stopMacro(m)
		end
	end
end)

----------------------------------------------------------------------
-- SAVE / LOAD (JSON)
-- Usa writefile/readfile/isfile (só existem em executores). Se não
-- existirem, o painel avisa e o resto continua funcionando.
----------------------------------------------------------------------
local HttpService = game:GetService("HttpService")
local SAVE_FOLDER, SAVE_FILE = "SukyWare", "SukyWare/config.json"
local fsOK = typeof(writefile) == "function" and typeof(readfile) == "function" and typeof(isfile) == "function"
local autoLoadOn = false
local autoToggle

local function c3(c)
	return { c.R, c.G, c.B }
end

local function readSave()
	if not fsOK or not isfile(SAVE_FILE) then
		return nil
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(readfile(SAVE_FILE))
	end)
	return (ok and type(data) == "table") and data or nil
end

local function writeSave(data)
	if not fsOK then
		notify("Seu executor não suporta writefile/readfile")
		return false
	end
	local ok = pcall(function()
		if typeof(isfolder) == "function" and typeof(makefolder) == "function" and not isfolder(SAVE_FOLDER) then
			makefolder(SAVE_FOLDER)
		end
		writefile(SAVE_FILE, HttpService:JSONEncode(data))
	end)
	return ok
end

local function saveConfig()
	local values = {}
	for _, key in ipairs(saveOrder) do
		local e = saveRegistry[key]
		local v = e.w.Get()
		if e.kind == "keybind" then
			v = v.Name
		end
		values[key] = v
	end
	local data = {
		version = 1,
		autoLoad = autoLoadOn,
		accent = c3(ACCENT),
		trailColor = c3(cfg.soru.trailColor),
		values = values,
	}
	notify(writeSave(data) and "Config salva" or "Falha ao salvar config")
end

local function loadConfig(manual)
	local data = readSave()
	if not data then
		if manual then
			notify("Nenhuma config salva")
		end
		return false
	end
	if data.accent then
		setAccent(Color3.new(unpack(data.accent)))
	end
	if data.trailColor then
		cfg.soru.trailColor = Color3.new(unpack(data.trailColor))
	end
	for _, key in ipairs(saveOrder) do
		local e = saveRegistry[key]
		local v = data.values and data.values[key]
		if v ~= nil then
			pcall(function()
				if e.kind == "keybind" then
					e.w.Set(Enum.KeyCode[v], true)
				elseif e.kind == "dropdown" or e.kind == "input" then
					e.w.Set(v, true)
				else
					e.w.Set(v) -- toggle/slider já disparam o callback
				end
			end)
		end
	end
	notify("Config carregada")
	return true
end

-- ABRIR / FECHAR COM ANIMAÇÃO -----------------------------------------
-- (declaradas no topo como locals; aqui só atribuímos)
function showPanel()
	closeDropdown()
	bubble.Visible = false
	panelOpen = true
	window.Visible = true

	if not anim.enabled then
		closing = false
		window.GroupTransparency = 0
		uiScale.Scale = userScale
		return
	end

	-- Se estava no meio do fechamento, continua de onde está (sem "pulo")
	if not closing then
		window.GroupTransparency = 1
		uiScale.Scale = userScale * anim.openScale
	end
	closing = false
	tweenEx(window, anim.openTime, { GroupTransparency = 0 }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	tweenEx(uiScale, anim.openTime, { Scale = userScale }, EASING[anim.style] or Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
end

function hidePanel()
	closeDropdown()
	panelOpen = false
	hideToken += 1
	local myToken = hideToken

	local function finish()
		closing = false
		window.Visible = false
		window.GroupTransparency = 0
		uiScale.Scale = userScale
		bubbleScale.Scale = 0
		bubble.Visible = true
		tween(bubbleScale, 0.2, { Scale = 1 })
	end

	if not anim.enabled then
		finish()
		return
	end

	closing = true
	tweenEx(window, anim.openTime, { GroupTransparency = 1 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
	local tw = tweenEx(uiScale, anim.openTime, { Scale = userScale * anim.openScale }, EASING[anim.style] or Enum.EasingStyle.Quad, Enum.EasingDirection.In)
	tw.Completed:Connect(function(state)
		-- só finaliza se esse fechamento terminou de verdade e ninguém reabriu no meio
		if state == Enum.PlaybackState.Completed and myToken == hideToken and closing and not panelOpen then
			finish()
		end
	end)
end

-- Arrastar a bolinha; um clique simples (sem arrastar) reabre o painel
do
	local dragging, moved = false, false
	local dragStart, startAbs

	bubble.InputBegan:Connect(function(input)
		if isPointer(input) then
			dragging, moved = true, false
			dragStart = input.Position
			startAbs = bubble.AbsolutePosition
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			if d.Magnitude > 6 then
				moved = true
			end
			if moved then
				local screen = gui.AbsoluteSize
				local x = math.clamp(startAbs.X + d.X, 0, screen.X - BUBBLE_SIZE)
				local y = math.clamp(startAbs.Y + d.Y, 0, screen.Y - BUBBLE_SIZE)
				bubble.Position = UDim2.fromOffset(x, y)
			end
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if dragging and isPointer(input) then
			dragging = false
			if not moved then
				showPanel()
			end
		end
	end)
end

closeButton.MouseButton1Click:Connect(hidePanel)

UserInputService.InputBegan:Connect(function(input, processed)
	if processed then
		return
	end
	if input.KeyCode == toggleKey then
		if panelOpen then
			hidePanel()
		else
			showPanel()
		end
	end
end)
